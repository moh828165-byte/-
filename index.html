// ========================================================== // 🛒 الشفرة البرمجية الشاملة لتطبيق "مول الصعيدي" (بدون أخطاء) // ========================================================== // 1. دالة حفظ الفاتورة وإتمام البيع مع مزامنة المخزن لحظياً async function checkoutInvoice(customerData, cartItems, discountValue, paidValue) { try { if (!cartItems || cartItems.length === 0) { alert("⚠️ السلة فارغة، لا يمكن إتمام الفاتورة."); return false; } let subtotal = cartItems.reduce((sum, item) => sum + (item.price * item.quantity), 0); let total = subtotal - (discountValue || 0); let remaining = total - (paidValue || total); let newInvoice = { id: "INV_" + Math.random().toString(36).substr(2, 6).toUpperCase(), date: new Date().toISOString().slice(0, 10) + " " + new Date().toLocaleTimeString('ar-EG'), customerName: customerData.name || "عميل نقدي", customerPhone: customerData.phone || "بدون هاتف", items: cartItems, subtotal: subtotal, discount: discountValue || 0, total: total, paid: paidValue || total, remaining: remaining }; // حفظ الفاتورة في سحابة فايربيส if (typeof db !== 'undefined') { await db.collection("invoices").add(newInvoice); // خصم الكميات المباعة من المخزون تلقائياً لكل صنف لضمان دقة المخزن for (let item of cartItems) { let productRef = db.collection("products").doc(item.id); let doc = await productRef.get(); if (doc.exists) { let currentStock = doc.data().stock || 0; let updatedStock = Math.max(0, currentStock - item.quantity); await productRef.update({ stock: updatedStock }); } } } alert("✅ تمت عملية البيع وحفظ الفاتورة وتحديث المخزن بنجاح تام!"); return true; } catch (error) { console.error("خطأ أثناء إتمام الفاتورة:", error); alert("❌ حدث خطأ أثناء الحفظ، تأكد من الاتصال بالإنترنت."); return false; } } // 2. تصدير تقرير الـ CSV المفصل (أصناف وكميات لكل فاتورة) function exportDetailedCSV() { let csvContent = "\uFEFFرقم الفاتورة,التاريخ,اسم العميل,رقم الهاتف,الصنف,الكمية,سعر القطعة,إجمالي الصنف,إجمالي الفاتورة\n"; if (typeof invoices !== 'undefined' && Array.isArray(invoices)) { invoices.forEach(inv => { if (inv.items && inv.items.length > 0) { inv.items.forEach(item => { let row = [ `"${inv.id || ''}"`, `"${inv.date || ''}"`, `"${inv.customerName || 'عميل نقدي'}"`, `"${inv.customerPhone || 'بدون هاتف'}"`, `"${item.name || ''}"`, item.quantity || 0, item.price || 0, (item.quantity * item.price) || 0, inv.total || 0 ].join(","); csvContent += row + "\n"; }); } }); } let blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' }); let url = URL.createObjectURL(blob); let a = document.createElement('a'); a.href = url; a.download = `Saidi_Mall_Report_${new Date().toISOString().slice(0,10)}.csv`; document.body.appendChild(a); a.click(); document.body.removeChild(a); } // 3. إصلاح الطباعة بشكل نظيف تماماً للموبايل والمتصفحات function printInvoice(invoiceElementId) { let element = document.getElementById(invoiceElementId); if (!element) { alert("عذراً، محتوى الفاتورة غير موجود للطباعة."); return; } let printContents = element.innerHTML; let originalContents = document.body.innerHTML; document.body.innerHTML = `

🛒 مول الصعيدي - فاتورة مبيعات

${printContents}
`; window.print(); document.body.innerHTML = originalContents; window.location.reload(); } // 4. خيارات مشاركة الواتساب (تقرير نصي منظم بدقة) function shareAsTextWhatsApp(invoice) { let itemsText = invoice.items.map(i => `▪️ ${i.name} (عدد: ${i.quantity}) - ${i.price * i.quantity} ج.م`).join("\n"); let message = `🛒 *مول الصعيدي* - فاتورة مبيعات\n` + `---------------------------\n` + `📄 *رقم الفاتورة:* ${invoice.id}\n` + `📅 *التاريخ:* ${invoice.date}\n` + `👤 *العميل:* ${invoice.customerName || 'عميل نقدي'}\n` + `---------------------------\n` + `📦 *الأصناف المباعة:*\n${itemsText}\n` + `---------------------------\n` + `💰 *إجمالي الأصناف:* ${invoice.subtotal || invoice.total} ج.م\n` + `🏷️ *الخصم:* ${invoice.discount || 0} ج.م\n` + `💵 *الإجمالي النهائي:* ${invoice.total} ج.م\n` + `💳 *المدفوع:* ${invoice.paid || invoice.total} ج.م\n` + `🔄 *المتبقي:* ${invoice.remaining || 0} ج.م\n` + `---------------------------\n` + `شكراً لتعاملكم معنا في مول الصعيدي! 🙏`; let encodedMessage = encodeURIComponent(message); window.open(`https://api.whatsapp.com/send?text=${encodedMessage}`, '_blank'); } // 5. مشاركة الفاتورة كصورة احترافية عبر الواتساب function shareAsImageWhatsApp(elementId) { const invoiceElement = document.getElementById(elementId); if (!invoiceElement) { alert("عذراً، عنصر الفاتورة غير موجود."); return; } if (typeof html2canvas === 'undefined') { alert("مكتبة تحويل الصور غير متوفرة."); return; } html2canvas(invoiceElement, { scale: 2 }).then(canvas => { canvas.toBlob(blob => { let file = new File([blob], "invoice.png", { type: "image/png" }); if (navigator.canShare && navigator.canShare({ files: [file] })) { navigator.share({ files: [file], title: 'فاتورة مول الصعيدي', text: 'تفاصيل الفاتورة من مول الصعيدي' }).catch(err => console.log('خطأ في المشاركة:', err)); } else { let link = document.createElement('a'); link.href = URL.createObjectURL(blob); link.download = 'Invoice_Saidi_Mall.png'; link.click(); alert("تم حفظ الفاتورة كصورة على جهازك، يمكنك إرسالها الآن عبر واتساب!"); window.open(`https://api.whatsapp.com/send?text=${encodeURIComponent("إليك فاتورة مبيعات مول الصعيدي:")}`, '_blank'); } }); }); }
